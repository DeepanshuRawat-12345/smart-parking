Invalid credentials

Error loading data

`; } } // Load active visitors async function loadActiveVisitors() { const tableBody = document.getElementById('activeVisitorsTable'); try { const response = await fetch(`${API_BASE}/requests`); const requests = await response.json(); const active = requests.filter(r => r.verified && !r.exit_time); if (active.length === 0) { tableBody.innerHTML = `

No active visitors

`; return; } tableBody.innerHTML = active.map(r => ` #${r.id} ${r.visitor_name} ${r.flat_number} ${r.assigned_slot || '-'} ${r.entry_time ? new Date(r.entry_time).toLocaleString() : '-'} `).join(''); } catch (error) { tableBody.innerHTML = `

Error loading data

`; } } // Verify entry async function verifyEntry() { const requestId = document.getElementById('entryRequestId').value; if (!requestId) { showAlert('entryAlert', 'Please enter a request ID'); return; } try { const response = await fetch( `${API_BASE}/verify-entry?request_id=${requestId}`, { method: 'POST' } ); const data = await response.json(); if (data.slot) { document.getElementById('entryResult').classList.remove('hidden'); if (!requestId) { showAlert('exitAlert', 'Please enter a request ID'); return; } try { const response = await fetch( `${API_BASE}/exit?request_id=${requestId}`, { method: 'POST' } ); const data = await response.json(); if (data.slot_freed) { document.getElementById('exitResult').classList.remove('hidden'); document.getElementById('freedSlot').textContent = data.slot_freed; document.getElementById('duration').textContent = data.duration || '-'; } } // Image preview document.getElementById('plateImage').addEventListener('change', function(e) { const file = e.target.files[0]; if (file) { const reader = new FileReader(); reader.onload = function(e) { document.getElementById('previewImg').src = e.target.result; const data = await response.json(); document.getElementById('ocrResult').classList.remove('hidden'); document.getElementById('detectedPlate').textContent = data.detected_plate || '-'; document.getElementById('residentName').textContent = data.resident || data.message || 'Not found';
